{% comment %} Slider component - range input (requires custom JS for styling) Usage: Note: Basecoat doesn't have a slider class. Use 'input' class and add JS: const updateSlider = (el) => { const min = parseFloat(el.min || 0); const max = parseFloat(el.max || 100); const value = parseFloat(el.value); const percent = (max === min) ? 0 : ((value - min) / (max - min)) * 100; el.style.setProperty('--slider-value', `${percent}%`); }; {% endcomment %}
